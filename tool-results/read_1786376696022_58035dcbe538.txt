     1→'use client';
     2→
     3→import { useEffect, useState, useCallback } from 'react';
     4→import { Plus, Search, Pencil, Eye, Trash2, X } from 'lucide-react';
     5→import { Card, CardContent } from '@/components/ui/card';
     6→import { Button } from '@/components/ui/button';
     7→import { Input } from '@/components/ui/input';
     8→import { Label } from '@/components/ui/label';
     9→import { Textarea } from '@/components/ui/textarea';
    10→import { Badge } from '@/components/ui/badge';
    11→import { Skeleton } from '@/components/ui/skeleton';
    12→import {
    13→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    14→} from '@/components/ui/table';
    15→import {
    16→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter,
    17→} from '@/components/ui/dialog';
    18→import {
    19→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    20→} from '@/components/ui/select';
    21→import { toast } from 'sonner';
    22→import type { CargoItem, LiftCategory, CommodityType, CargoStatus, Location, Project } from '@/types/wms';
    23→
    24→const statusStyles: Record<CargoStatus, string> = {
    25→  IN_YARD: 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20',
    26→  IN_TRANSIT: 'bg-amber-500/10 text-amber-400 border-amber-500/20',
    27→  DISPATCHED: 'bg-slate-500/10 text-slate-400 border-slate-500/20',
    28→  RECEIVED: 'bg-cyan-500/10 text-cyan-400 border-cyan-500/20',
    29→  IN_WAREHOUSE: 'bg-teal-500/10 text-teal-400 border-teal-500/20',
    30→  DELIVERED: 'bg-slate-400/10 text-slate-500 border-slate-400/20',
    31→};
    32→
    33→const categoryStyles: Record<LiftCategory, string> = {
    34→  HEAVY_LIFT: 'bg-red-500/10 text-red-400 border-red-500/20',
    35→  OVERSIZE: 'bg-orange-500/10 text-orange-400 border-orange-500/20',
    36→  STANDARD: 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20',
    37→  PROJECT_CARGO: 'bg-amber-500/10 text-amber-400 border-amber-500/20',
    38→};
    39→
    40→interface CargoListResponse {
    41→  items: CargoItem[];
    42→  total: number;
    43→  page: number;
    44→  totalPages: number;
    45→}
    46→
    47→const emptyForm = {
    48→  description: '', weight: '', length: '', width: '', height: '',
    49→  liftCategory: '' as string, commodityType: '' as string,
    50→  specialHandling: '', clientName: '', poReference: '', blReference: '',
    51→  centerOfGravity: '', liftingPoints: '', projectId: '',
    52→};
    53→
    54→export function CargoPage() {
    55→  const [cargo, setCargo] = useState<CargoItem[]>([]);
    56→  const [totalPages, setTotalPages] = useState(1);
    57→  const [page, setPage] = useState(1);
    58→  const [loading, setLoading] = useState(true);
    59→  const [search, setSearch] = useState('');
    60→  const [statusFilter, setStatusFilter] = useState('');
    61→  const [categoryFilter, setCategoryFilter] = useState('');
    62→  const [commodityFilter, setCommodityFilter] = useState('');
    63→  const [showAdd, setShowAdd] = useState(false);
    64→  const [editing, setEditing] = useState<CargoItem | null>(null);
    65→  const [viewing, setViewing] = useState<CargoItem | null>(null);
    66→  const [deleting, setDeleting] = useState<CargoItem | null>(null);
    67→  const [form, setForm] = useState(emptyForm);
    68→  const [submitting, setSubmitting] = useState(false);
    69→  const [locations, setLocations] = useState<Location[]>([]);
    70→  const [projects, setProjects] = useState<Project[]>([]);
    71→
    72→  const fetchCargo = useCallback(async () => {
    73→    setLoading(true);
    74→    const params = new URLSearchParams({
    75→      page: String(page),
    76→      limit: '10',
    77→      ...(search && { search }),
    78→      ...(statusFilter && { status: statusFilter }),
    79→      ...(categoryFilter && { liftCategory: categoryFilter }),
    80→      ...(commodityFilter && { commodityType: commodityFilter }),
    81→    });
    82→    try {
    83→      const res = await fetch(`/api/cargo?${params}`);
    84→      const data: CargoListResponse = await res.json();
    85→      setCargo(data.items);
    86→      setTotalPages(data.totalPages);
    87→    } catch {
    88→      toast.error('Failed to fetch cargo items');
    89→    } finally {
    90→      setLoading(false);
    91→    }
    92→  }, [page, search, statusFilter, categoryFilter, commodityFilter]);
    93→
    94→  const fetchLookups = useCallback(async () => {
    95→    try {
    96→      const [locRes, projRes] = await Promise.all([
    97→        fetch('/api/locations?limit=100'),
    98→        fetch('/api/projects?limit=100'),
    99→      ]);
   100→      const locData = await locRes.json();
   101→      const projData = await projRes.json();
   102→      setLocations(locData.items || []);
   103→      setProjects(projData.items || []);
   104→    } catch {
   105→      // silent
   106→    }
   107→  }, []);
   108→
   109→  useEffect(() => { fetchCargo(); }, [fetchCargo]);
   110→  useEffect(() => { fetchLookups(); }, [fetchLookups]);
   111→  useEffect(() => { setPage(1); }, [search, statusFilter, categoryFilter, commodityFilter]);
   112→
   113→  const handleSubmit = async () => {
   114→    setSubmitting(true);
   115→    const payload: Record<string, unknown> = {
   116→      description: form.description,
   117→      weight: parseFloat(form.weight) || 0,
   118→      length: parseFloat(form.length) || 0,
   119→      width: parseFloat(form.width) || 0,
   120→      height: parseFloat(form.height) || 0,
   121→      liftCategory: form.liftCategory,
   122→      commodityType: form.commodityType,
   123→      specialHandling: form.specialHandling || null,
   124→      clientName: form.clientName || null,
   125→      poReference: form.poReference || null,
   126→      blReference: form.blReference || null,
   127→      centerOfGravity: form.centerOfGravity || null,
   128→      liftingPoints: form.liftingPoints ? parseInt(form.liftingPoints) : null,
   129→      projectId: form.projectId || null,
   130→    };
   131→
   132→    try {
   133→      const url = editing ? `/api/cargo/${editing.id}` : '/api/cargo';
   134→      const method = editing ? 'PUT' : 'POST';
   135→      const res = await fetch(url, {
   136→        method,
   137→        headers: { 'Content-Type': 'application/json' },
   138→        body: JSON.stringify(payload),
   139→      });
   140→      if (!res.ok) {
   141→        const err = await res.json();
   142→        throw new Error(err.error || 'Failed to save');
   143→      }
   144→      toast.success(editing ? 'Cargo updated' : 'Cargo created');
   145→      setShowAdd(false);
   146→      setEditing(null);
   147→      setForm(emptyForm);
   148→      fetchCargo();
   149→    } catch (e: unknown) {
   150→      toast.error(e instanceof Error ? e.message : 'Failed to save');
   151→    } finally {
   152→      setSubmitting(false);
   153→    }
   154→  };
   155→
   156→  const handleDelete = async () => {
   157→    if (!deleting) return;
   158→    try {
   159→      const res = await fetch(`/api/cargo/${deleting.id}`, { method: 'DELETE' });
   160→      if (!res.ok) throw new Error('Failed to delete');
   161→      toast.success('Cargo deleted');
   162→      setDeleting(null);
   163→      fetchCargo();
   164→    } catch {
   165→      toast.error('Failed to delete cargo');
   166→    }
   167→  };
   168→
   169→  const openEdit = (item: CargoItem) => {
   170→    setEditing(item);
   171→    setForm({
   172→      description: item.description,
   173→      weight: String(item.weight),
   174→      length: String(item.length),
   175→      width: String(item.width),
   176→      height: String(item.height),
   177→      liftCategory: item.liftCategory,
   178→      commodityType: item.commodityType,
   179→      specialHandling: item.specialHandling || '',
   180→      clientName: item.clientName || '',
   181→      poReference: item.poReference || '',
   182→      blReference: item.blReference || '',
   183→      centerOfGravity: item.centerOfGravity || '',
   184→      liftingPoints: item.liftingPoints ? String(item.liftingPoints) : '',
   185→      projectId: item.projectId || '',
   186→    });
   187→  };
   188→
   189→  return (
   190→    <div className="space-y-6">
   191→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   192→        <div>
   193→          <h1 className="text-2xl font-bold text-slate-100">Cargo Items</h1>
   194→          <p className="text-sm text-slate-500 mt-1">Manage all cargo in the warehouse</p>
   195→        </div>
   196→        <Button
   197→          onClick={() => { setForm(emptyForm); setShowAdd(true); }}
   198→          className="bg-amber-500 hover:bg-amber-600 text-slate-900 font-medium"
   199→        >
   200→          <Plus className="h-4 w-4 mr-2" /> Add Cargo
   201→        </Button>
   202→      </div>
   203→
   204→      {/* Filters */}
   205→      <Card className="border-slate-800 bg-slate-900/50">
   206→        <CardContent className="p-4">
   207→          <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
   208→            <div className="relative">
   209→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-slate-500" />
   210→              <Input
   211→                placeholder="Search code, description..."
   212→                value={search}
   213→                onChange={(e) => setSearch(e.target.value)}
   214→                className="border-slate-700 bg-slate-800 pl-9 text-slate-200 placeholder:text-slate-600"
   215→              />
   216→            </div>
   217→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   218→              <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-300">
   219→                <SelectValue placeholder="All Statuses" />
   220→              </SelectTrigger>
   221→              <SelectContent className="border-slate-700 bg-slate-800">
   222→                {['IN_TRANSIT', 'RECEIVED', 'IN_YARD', 'IN_WAREHOUSE', 'DISPATCHED', 'DELIVERED'].map((s) => (
   223→                  <SelectItem key={s} value={s} className="text-slate-300 focus:bg-slate-700 focus:text-slate-100">
   224→                    {s.replace(/_/g, ' ')}
   225→                  </SelectItem>
   226→                ))}
   227→              </SelectContent>
   228→            </Select>
   229→            <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   230→              <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-300">
   231→                <SelectValue placeholder="All Categories" />
   232→              </SelectTrigger>
   233→              <SelectContent className="border-slate-700 bg-slate-800">
   234→                {['HEAVY_LIFT', 'OVERSIZE', 'STANDARD', 'PROJECT_CARGO'].map((c) => (
   235→                  <SelectItem key={c} value={c} className="text-slate-300 focus:bg-slate-700 focus:text-slate-100">
   236→                    {c.replace(/_/g, ' ')}
   237→                  </SelectItem>
   238→                ))}
   239→              </SelectContent>
   240→            </Select>
   241→            <Select value={commodityFilter} onValueChange={setCommodityFilter}>
   242→              <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-300">
   243→                <SelectValue placeholder="All Commodities" />
   244→              </SelectTrigger>
   245→              <SelectContent className="border-slate-700 bg-slate-800">
   246→                {['GENERAL', 'MACHINERY', 'STEEL', 'EQUIPMENT', 'MODULE'].map((c) => (
   247→                  <SelectItem key={c} value={c} className="text-slate-300 focus:bg-slate-700 focus:text-slate-100">
   248→                    {c}
   249→                  </SelectItem>
   250→                ))}
   251→              </SelectContent>
   252→            </Select>
   253→          </div>
   254→        </CardContent>
   255→      </Card>
   256→
   257→      {/* Table */}
   258→      <Card className="border-slate-800 bg-slate-900/50">
   259→        <CardContent className="p-0">
   260→          <div className="max-h-96 overflow-y-auto">
   261→            <Table>
   262→              <TableHeader>
   263→                <TableRow className="border-slate-800 hover:bg-transparent">
   264→                  <TableHead className="text-xs text-slate-500">Code</TableHead>
   265→                  <TableHead className="text-xs text-slate-500">Description</TableHead>
   266→                  <TableHead className="text-xs text-slate-500 hidden md:table-cell">Weight (kg)</TableHead>
   267→                  <TableHead className="text-xs text-slate-500 hidden lg:table-cell">Dims (L×W×H)</TableHead>
   268→                  <TableHead className="text-xs text-slate-500 hidden sm:table-cell">Category</TableHead>
   269→                  <TableHead className="text-xs text-slate-500">Status</TableHead>
   270→                  <TableHead className="text-xs text-slate-500 hidden lg:table-cell">Location</TableHead>
   271→                  <TableHead className="text-xs text-slate-500 hidden xl:table-cell">Project</TableHead>
   272→                  <TableHead className="text-xs text-slate-500 text-right">Actions</TableHead>
   273→                </TableRow>
   274→              </TableHeader>
   275→              <TableBody>
   276→                {loading
   277→                  ? Array.from({ length: 5 }).map((_, i) => (
   278→                      <TableRow key={i} className="border-slate-800 hover:bg-transparent">
   279→                        {Array.from({ length: 9 }).map((_, j) => (
   280→                          <TableCell key={j} className="py-3"><Skeleton className="h-4 w-16 bg-slate-800" /></TableCell>
   281→                        ))}
   282→                      </TableRow>
   283→                    ))
   284→                  : cargo.length === 0
   285→                    ? (
   286→                      <TableRow className="border-slate-800 hover:bg-transparent">
   287→                        <TableCell colSpan={9} className="text-center py-8 text-slate-500">
   288→                          No cargo items found
   289→                        </TableCell>
   290→                      </TableRow>
   291→                    )
   292→                    : cargo.map((item) => (
   293→                      <TableRow key={item.id} className="border-slate-800 hover:bg-slate-800/50">
   294→                        <TableCell className="py-3 text-xs font-mono font-medium text-amber-400/80">{item.cargoCode}</TableCell>
   295→                        <TableCell className="py-3 text-xs text-slate-300 max-w-[200px] truncate">{item.description}</TableCell>
   296→                        <TableCell className="py-3 text-xs text-slate-400 hidden md:table-cell">{item.weight.toLocaleString()}</TableCell>
   297→                        <TableCell className="py-3 text-xs text-slate-400 hidden lg:table-cell font-mono">
   298→                          {item.length}×{item.width}×{item.height}
   299→                        </TableCell>
   300→                        <TableCell className="py-3 hidden sm:table-cell">
   301→                          <Badge variant="outline" className={`text-[10px] ${categoryStyles[item.liftCategory]}`}>{item.liftCategory.replace(/_/g, ' ')}</Badge>
   302→                        </TableCell>
   303→                        <TableCell className="py-3">
   304→                          <Badge variant="outline" className={`text-[10px] ${statusStyles[item.status]}`}>{item.status.replace(/_/g, ' ')}</Badge>
   305→                        </TableCell>
   306→                        <TableCell className="py-3 text-xs text-slate-400 hidden lg:table-cell">{item.location?.code || '—'}</TableCell>
   307→                        <TableCell className="py-3 text-xs text-slate-400 hidden xl:table-cell max-w-[120px] truncate">{item.project?.name || '—'}</TableCell>
   308→                        <TableCell className="py-3 text-right">
   309→                          <div className="flex items-center justify-end gap-1">
   310→                            <Button variant="ghost" size="icon" className="h-7 w-7 text-slate-500 hover:text-cyan-400" onClick={() => setViewing(item)}>
   311→                              <Eye className="h-3.5 w-3.5" />
   312→                            </Button>
   313→                            <Button variant="ghost" size="icon" className="h-7 w-7 text-slate-500 hover:text-amber-400" onClick={() => openEdit(item)}>
   314→                              <Pencil className="h-3.5 w-3.5" />
   315→                            </Button>
   316→                            <Button variant="ghost" size="icon" className="h-7 w-7 text-slate-500 hover:text-red-400" onClick={() => setDeleting(item)}>
   317→                              <Trash2 className="h-3.5 w-3.5" />
   318→                            </Button>
   319→                          </div>
   320→                        </TableCell>
   321→                      </TableRow>
   322→                    ))}
   323→              </TableBody>
   324→            </Table>
   325→          </div>
   326→        </CardContent>
   327→      </Card>
   328→
   329→      {/* Pagination */}
   330→      {totalPages > 1 && (
   331→        <div className="flex items-center justify-center gap-2">
   332→          <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => setPage(page - 1)}
   333→            className="border-slate-700 bg-slate-800 text-slate-300 hover:bg-slate-700">
   334→            Previous
   335→          </Button>
   336→          <span className="text-xs text-slate-500">Page {page} of {totalPages}</span>
   337→          <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => setPage(page + 1)}
   338→            className="border-slate-700 bg-slate-800 text-slate-300 hover:bg-slate-700">
   339→            Next
   340→          </Button>
   341→        </div>
   342→      )}
   343→
   344→      {/* Add/Edit Dialog */}
   345→      <Dialog open={showAdd || !!editing} onOpenChange={(open) => { if (!open) { setShowAdd(false); setEditing(null); setForm(emptyForm); } }}>
   346→        <DialogContent className="border-slate-700 bg-slate-900 max-h-[90vh] overflow-y-auto max-w-2xl">
   347→          <DialogHeader>
   348→            <DialogTitle className="text-slate-100">{editing ? 'Edit Cargo' : 'Add New Cargo'}</DialogTitle>
   349→          </DialogHeader>
   350→          <div className="grid gap-4 py-4 sm:grid-cols-2">
   351→            <div className="sm:col-span-2">
   352→              <Label className="text-slate-400">Description *</Label>
   353→              <Input value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })}
   354→                className="border-slate-700 bg-slate-800 text-slate-200 mt-1" />
   355→            </div>
   356→            <div>
   357→              <Label className="text-slate-400">Weight (kg) *</Label>
   358→              <Input type="number" value={form.weight} onChange={(e) => setForm({ ...form, weight: e.target.value })}
   359→                className="border-slate-700 bg-slate-800 text-slate-200 mt-1" />
   360→            </div>
   361→            <div>
   362→              <Label className="text-slate-400">Dimensions (m)</Label>
   363→              <div className="grid grid-cols-3 gap-2 mt-1">
   364→                <Input placeholder="L" type="number" value={form.length} onChange={(e) => setForm({ ...form, length: e.target.value })}
   365→                  className="border-slate-700 bg-slate-800 text-slate-200 text-center" />
   366→                <Input placeholder="W" type="number" value={form.width} onChange={(e) => setForm({ ...form, width: e.target.value })}
   367→                  className="border-slate-700 bg-slate-800 text-slate-200 text-center" />
   368→                <Input placeholder="H" type="number" value={form.height} onChange={(e) => setForm({ ...form, height: e.target.value })}
   369→                  className="border-slate-700 bg-slate-800 text-slate-200 text-center" />
   370→              </div>
   371→            </div>
   372→            <div>
   373→              <Label className="text-slate-400">Lift Category *</Label>
   374→              <Select value={form.liftCategory} onValueChange={(v) => setForm({ ...form, liftCategory: v })}>
   375→                <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-200 mt-1"><SelectValue placeholder="Select..." /></SelectTrigger>
   376→                <SelectContent className="border-slate-700 bg-slate-800">
   377→                  {(['HEAVY_LIFT', 'OVERSIZE', 'STANDARD', 'PROJECT_CARGO'] as LiftCategory[]).map((c) => (
   378→                    <SelectItem key={c} value={c} className="text-slate-200 focus:bg-slate-700">{c.replace(/_/g, ' ')}</SelectItem>
   379→                  ))}
   380→                </SelectContent>
   381→              </Select>
   382→            </div>
   383→            <div>
   384→              <Label className="text-slate-400">Commodity Type *</Label>
   385→              <Select value={form.commodityType} onValueChange={(v) => setForm({ ...form, commodityType: v })}>
   386→                <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-200 mt-1"><SelectValue placeholder="Select..." /></SelectTrigger>
   387→                <SelectContent className="border-slate-700 bg-slate-800">
   388→                  {(['GENERAL', 'MACHINERY', 'STEEL', 'EQUIPMENT', 'MODULE'] as CommodityType[]).map((c) => (
   389→                    <SelectItem key={c} value={c} className="text-slate-200 focus:bg-slate-700">{c}</SelectItem>
   390→                  ))}
   391→                </SelectContent>
   392→              </Select>
   393→            </div>
   394→            <div>
   395→              <Label className="text-slate-400">Project</Label>
   396→              <Select value={form.projectId} onValueChange={(v) => setForm({ ...form, projectId: v })}>
   397→                <SelectTrigger className="border-slate-700 bg-slate-800 text-slate-200 mt-1"><SelectValue placeholder="None" /></SelectTrigger>
   398→                <SelectContent className="border-slate-700 bg-slate-800">
   399→                  <SelectItem value="_none" className="text-slate-200 focus:bg-slate-700">None</SelectItem>
   400→                  {projects.map((p) => (
   401→                    <SelectItem key={p.id} value={p.id} className="text-slate-200 focus:bg-slate-700">{p.name}</SelectItem>
   402→                  ))}
   403→                </SelectContent>
   404→              </Select>
   405→            </div>
   406→            <div>
   407→              <Label className="text-slate-400">Client Name</Label>
   408→              <Input value={form.clientName} onChange={(e) => setForm({ ...form, clientName: e.target.value })}
   409→                className="border-slate-700 bg-slate-800 text-slate-200 mt-1" />
   410→            </div>
   411→            <div>
   412→              <Label className="text-slate-400">PO Reference</Label>
   413→              <Input value={form.poReference} onChange={(e) => setForm({ ...form, poReference: e.target.value })}
   414→                className="border-slate-700 bg-slate-800 text-slate-200 mt-1" />
   415→            </div>
   416→            <div>
   417→              <Label className="text-slate-400">BL Reference</Label>
   418→              <Input value={form.blReference} onChange={(e) => setForm({ ...form, blReference: e.target.value })}
   419→                className="border-slate-700 bg-slate-800 text-slate-200 mt-1" />
   420→            </div>
   421→            <div>
   422→              <Label className="text-slate-400">Center of Gravity</Label>
   423→              <Input value={form.centerOfGravity} onChange={(e) => setForm({ ...form, centerOfGravity: e.target.value })}
   424→                className="border-slate-700 bg-slate-800 text-slate-200 mt-1" placeholder="e.g. 5.2m from base" />
   425→            </div>
   426→            <div>
   427→              <Label className="text-slate-400">Lifting Points</Label>
   428→              <Input type="number" value={form.liftingPoints} onChange={(e) => setForm({ ...form, liftingPoints: e.target.value })}
   429→                className="border-slate-700 bg-slate-800 text-slate-200 mt-1" placeholder="Number of points" />
   430→            </div>
   431→            <div className="sm:col-span-2">
   432→              <Label className="text-slate-400">Special Handling</Label>
